func NewShopHandler(logger *zap.Logger, shopService shop.ShopService) *ShopHandler {

		shopService: shopService,
	}
}

func (h *ShopHandler) SetQuestService(qs *appquest.Service) {

func (h *ShopHandler) BuyAmountItemClient(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	h.logger.Info("BuyAmountItemClient called", zap.Uint32("conn_id", ctx.ConnID), zap.Any("args", args))
	if len(args) < 2 {
		return nil, nil
	}

	// Safe argument parsing
	var shopSlotID int
	if v, ok := args[0].(float64); ok {
		shopSlotID = int(v)
	} else if v, ok := args[0].(string); ok {
		shopSlotID, _ = strconv.Atoi(v)
	}

	var amount int
	if v, ok := args[1].(float64); ok {
		amount = int(v)
	} else if v, ok := args[1].(string); ok {
		amount, _ = strconv.Atoi(v)
	}

	charID, _ := strconv.ParseInt(ctx.CharacterID, 10, 64)

	res, err := h.shopService.BuyItem(ctx.Context, charID, shopSlotID, amount)
	if err != nil {
		// Send error message as red message
		errorMsg := fmt.Sprintf("Mua thất bại: %s", err.Error())
		_ = ctx.Connection.SendCallback("onRedMsg", errorMsg)
		return nil, nil
	}

	// Send callbacks for UI update

	// Determine currency name for notification
	currencyName := getCurrencyName(res.CurrencyType)

	// Send success notification
	msg := fmt.Sprintf("Mua thành công x%d! Mất %d %s.", amount, res.Cost, currencyName)
	_ = ctx.Connection.SendCallback("onBlueMsg", msg)

	// Send callbacks for UI update

	// Dynamically update the correct currency field
	currencyKey := getCurrencyKey(res.CurrencyType)
	_ = ctx.Connection.SendCallback("onUPP", map[string]interface{}{currencyKey: res.NewMoney})

	// Use onAddTempDirect (effect) as requested
	// Args: itemData, slotIndex (0), itemData (slotData), curNum (1)
	// _ = ctx.Connection.SendCallback("onAddTempDirect", res.ItemData, 0, res.ItemData, 1)

	_ = ctx.Connection.SendCallback("onAddItem", res.ItemData)
	_ = ctx.Connection.SendCallback("onAddCharactorSlot", res.AddedItemDTO)

	h.checkQuestUpdates(ctx, charID)

	return true, nil
}

// getCurrencyKey returns the client property key for updating currency balance
func getCurrencyKey(currencyType int) string {
	switch currencyType {
	case 1:
		return "moneyBind"
	case 2:
		return "money"
	case 3:
		return "goldBind"
	case 4:
		return "gold"

	// Special currencies mapping
	case 20:
		return "guildContrib"
	case 61: // Shop Gold
		return "shopGold"
	case 64: // MC Beans
		return "mcBeans"

	// Case 0 for System Shop Point purchase (usually Honor)
	case 0:
		return "honor"

	default:
		return "money"
	}
}

func (h *ShopHandler) BuySystemItemMulti(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	h.logger.Info("BuySystemItemMulti called", zap.Uint32("conn_id", ctx.ConnID), zap.Any("args", args))
	if len(args) < 1 {
		return nil, nil
	}

	charID, _ := strconv.ParseInt(ctx.CharacterID, 10, 64)

	var purchases []shop.PurchaseRequest

	// Helper function to add a purchase to the list
	addPurchaseFunc := func(list *[]shop.PurchaseRequest, shopSlotIDVal, amountVal interface{}) {
		shopSlotID := parseIntVal(shopSlotIDVal)
		amount := parseIntVal(amountVal)

		if shopSlotID > 0 && amount > 0 {
			*list = append(*list, shop.PurchaseRequest{
				ShopSlotID: shopSlotID,
				Amount:     amount,
			})
		}
	}

	// Safe parsing of buyList
	// Case 1: Client sends a Map { "ShopSlotID": Amount, ... } directly
	if m, ok := args[0].(map[string]interface{}); ok {
		for k, v := range m {
			addPurchaseFunc(&purchases, k, v)
		}
	} else if list, ok := args[0].([]interface{}); ok {
		// Case 2: Client sends an Array (List)
		// Can contain: [{"2116":1}, 0] OR [{id:1, count:1}] OR [["ID",1]]
		for _, item := range list {
			if m, ok := item.(map[string]interface{}); ok {
				if val, ok := m["shopSlotId"]; ok {
					if amt, ok := m["amount"]; ok {
						addPurchaseFunc(&purchases, val, amt)
						continue
					}
				}
				if val, ok := m["id"]; ok {
					if amt, ok := m["count"]; ok {
						addPurchaseFunc(&purchases, val, amt)
						continue
					}
				}
				// If no standard keys, treat all keys as ID (Dynamic Map)
				// Example: {"3279":1, "3874":1}
				for k, v := range m {
					addPurchaseFunc(&purchases, k, v)
				}
				continue
			}
			if arr, ok := item.([]interface{}); ok && len(arr) >= 2 {
				addPurchaseFunc(&purchases, arr[0], arr[1])
			}
		}
	}

	if len(purchases) == 0 {
		h.logger.Warn("BuySystemItemMulti: no valid purchases found", zap.Uint32("conn_id", ctx.ConnID))
		return map[string]interface{}{
			"success": false,
			"items":   []interface{}{},
		}, nil
	}

	res, err := h.shopService.BuySystemItemMulti(ctx.Context, charID, purchases)
	if err != nil {
		h.logger.Error("BuySystemItemMulti failed", zap.Uint32("conn_id", ctx.ConnID), zap.Error(err))
		return map[string]interface{}{
			"success": false,
			"items":   []interface{}{},
		}, nil
	}

	// Send callbacks for UI update
	for _, itemData := range res.Items {
		_ = ctx.Connection.SendCallback("onAddItem", itemData)
	}
	for _, itemDTO := range res.AddedItemDTOs {
		_ = ctx.Connection.SendCallback("onAddCharactorSlot", itemDTO)
	}

	currencyKey := getCurrencyKey(res.CurrencyType)
	_ = ctx.Connection.SendCallback("onUPP", map[string]interface{}{currencyKey: res.NewMoney})

	h.checkQuestUpdates(ctx, charID)

	return map[string]interface{}{
		"success": res.Success,
		"items":   res.Items,
	}, nil
}

// Helper func to parse int from unknown type (float64, string, int)
func parseIntVal(v interface{}) int {
	switch val := v.(type) {
	case float64:
		return int(val)
	case string:
		i, _ := strconv.Atoi(val)
		return i
	case int:
		return val
	default:
		return 0
	}
}
