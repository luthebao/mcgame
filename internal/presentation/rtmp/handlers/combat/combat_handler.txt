func (h *CombatHandler) NpcFuncClick(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	if len(args) < 1 {
		return nil, pkgerrors.ErrInvalidArgs
	}
	var npcID int
	var passHash string
	foundID := false

	for _, arg := range args {
		if arg == nil {
			continue
		}
		if !foundID {
			switch v := arg.(type) {
			case float64:
				npcID = int(v)
				foundID = true
			case int:
				npcID = v
				foundID = true
			}
		} else {
			// Look for hash after ID
			if s, ok := arg.(string); ok {
				passHash = s
				break
			}
		}
	}

	if npcID == 0 {
		return nil, pkgerrors.ErrInvalidArgs
	}

	// 2. Look up NPC and Open Shop
	if h.gameData != nil {
		npc := h.gameData.GetNPC(npcID)
		if npc != nil {
			if npc.ShopID > 0 {
				shopID := int(npc.ShopID)
				h.logger.Info("NpcFuncClick opening shop", zap.Int("npc_id", npcID), zap.Int("shop_id", shopID))

				if shopID <= 7 {
					if err := ctx.Connection.SendCallbackSync("openShopById", shopID); err != nil {
						h.logger.Warn("Failed to send openShopById", zap.Error(err))
					}
				} else {
					if err := ctx.Connection.SendCallbackSync("onOpenShop", shopID); err != nil {
						h.logger.Warn("Failed to send onOpenShop", zap.Error(err))
					}
				}
				return map[string]interface{}{"success": true}, nil
			}

			// If not a shop, check for other types
			switch npc.Type {
			case 7: // NPC_TYPE_BANK (Or 5?)
				h.logger.Info("NpcFuncClick: opening Bank panel", zap.Int("npc_id", npcID))

				// If password hash is provided, verify it first (if not empty)
				if passHash != "" {
					accountID, err := uuid.Parse(ctx.AccountID)
					if err == nil {
						if acc, err := h.accountRepo.FindByID(ctx.Context, accountID); err == nil {
							// Check if stored secondary password is empty or matches
							// If checking against stored pass:
							if !acc.CheckSecondaryPassword(passHash) {
								h.logger.Warn("NpcFuncClick: Incorrect bank password", zap.String("accountID", ctx.AccountID))
								// Send callback indicating wrong password? Or just return success:false?
								// Sending "onNpcMsg" might be appropriate feedback
								ctx.Connection.SendCallback("onNpcMsg", npcID, "Mật khẩu rương không đúng!")
								return map[string]interface{}{"success": true}, nil
							}
						}
					}

					h.logger.Info("NpcFuncClick: sending setClientDP", zap.String("hash", passHash))
					if err := ctx.Connection.SendCallback("setClientDP", passHash); err != nil {
						h.logger.Error("Failed to send setClientDP", zap.Error(err))
					}
				}

				if err := ctx.Connection.SendCallback("initViewBankPanel"); err != nil {
					h.logger.Error("Failed to send initViewBankPanel", zap.Error(err))
				}
				return map[string]interface{}{"success": true}, nil
			case 5: // NPC_TYPE_AUCTION (Or 7?)
				h.logger.Info("NpcFuncClick: opening Auction panel", zap.Int("npc_id", npcID))
				data := map[string]interface{}{
					"flag":          true,
					"npcId":         npcID,
					"myAuctionList": map[string]interface{}{},
				}
				if err := ctx.Connection.SendCallback("onInitViewAuctionP", data); err != nil {
					h.logger.Error("Failed to send onInitViewAuctionP", zap.Error(err))
				}
				return map[string]interface{}{"success": true}, nil
			case 13: // CallBoard
				h.logger.Info("NpcFuncClick: CallBoard clicked", zap.Int("npc_id", npcID))
			}
		}
	}

	h.logger.Warn("NpcFuncClick called for NPC without shop or not found", zap.Int("npc_id", npcID))
	return map[string]interface{}{"success": false}, nil
}

func (h *CombatHandler) NpcScript(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	if len(args) < 1 {
		return nil, pkgerrors.ErrInvalidArgs
	}

	var funcID string
	switch v := args[0].(type) {
	case string:
		funcID = v
	case float64:
		funcID = fmt.Sprintf("%v", v)
	case int:
		funcID = fmt.Sprintf("%d", v)
	default:
		return nil, pkgerrors.ErrInvalidArgs
	}

	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	h.logger.Info("NPC script action triggered",
		zap.Int64("character_id", characterID),
		zap.String("func_id", funcID))

	// Handle specific function logics
	if strings.HasPrefix(funcID, "transport_") {
		parts := strings.Split(funcID, "_")
		if len(parts) == 5 {
			mapID, _ := strconv.Atoi(parts[1])
			cost, _ := strconv.Atoi(parts[2])

			// Get character to check/deduct money
			char, err := h.charService.GetByID(ctx.Context, characterID)
			if err != nil {
				h.logger.Error("Failed to get character", zap.Error(err))
				return nil, err
			}

			// Check cost (assumed Money/Silver)
			if int(char.MoneyBind) < cost {
				h.logger.Info("Not enough money", zap.Int64("charID", characterID))
				return map[string]interface{}{"success": false}, nil
			}

			// Deduct money
			char.MoneyBind -= int64(cost)

			// Notify client
			ctx.Connection.SendCallback("onUPP", map[string]interface{}{
				"moneyBind": char.MoneyBind,
			})
			ctx.Connection.SendCallback("onBlueMsg", fmt.Sprintf("Mất %d Ngân phiếu.", cost))

			if err := h.teleportToMap(ctx, characterID, mapID); err != nil {
				return nil, err
			}

			return map[string]interface{}{"success": true}, nil
		}
	}

	if strings.HasPrefix(funcID, "hlvt_next_floor_") {
		idStr := funcID[len("hlvt_next_floor_"):]
		id, _ := strconv.Atoi(idStr)

		// Mapping logic for 2293-2300
		// 2293->80, 2294->81, 2295->82, 2296->83, 2297->84, 2298->85, 2299->86
		// 2300 is floor 10, target next available or loop
		targetMap := 0
		if id >= 2293 && id <= 2299 {
			targetMap = 80 + (id - 2293)
		} else if id == 2300 {
			targetMap = 79 // Or next logical map
		}

		if targetMap > 0 {
			if err := h.teleportToMap(ctx, characterID, targetMap); err != nil {
				return nil, err
			}
		} else {
			ctx.Connection.SendCallback("onNpcMsg", id, "Tầng này chưa mở hoặc không có đường đi tiếp.")
		}
		return map[string]interface{}{"success": true}, nil

	if strings.HasPrefix(funcID, "hlvt_explore_more_") {

	if strings.HasPrefix(funcID, "hlvt_boss_current_") {

func (h *CombatHandler) HitNpc(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	if len(args) < 1 {
		return nil, pkgerrors.ErrInvalidArgs
	}

	var npcID int
	switch v := args[0].(type) {
	case float64:
		npcID = int(v)
	case int:
		npcID = v
	default:
		return nil, pkgerrors.ErrInvalidArgs
	}

	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	battle, err := h.combatService.StartPVEBattle(ctx.Context, characterID, npcID)
	if err != nil {
		return rtmp.ErrorToResponse(err), nil
	}

	if h.battleService != nil {
		h.battleService.SetTurnDeadline(battle.ID, fmt.Sprintf("%d", characterID))
	}

	ctx.Connection.SetBattleID(battle.ID)

	h.logger.Info("PVE battle initiated",
		zap.Int64("character_id", characterID),
		zap.Int("npc_id", npcID),
		zap.String("battle_id", battle.ID))

	battleDTO := battle.ToDTO()
	cList := battleDTO["cList"]
	battleData := map[string]interface{}{
		"battleFieldId": battle.ID,
		"nid":           npcID,
		"battleType":    int(combat.BattleTypePVE),
		"mapId":         battle.MapID,
		"battleData":    battleDTO,
		"cList":         cList,
		"guest":         false,
		"sneakFlag":     0,
	}
	if err := ctx.Connection.SendCallback("onBattleStart", battleData); err != nil {
		h.logger.Error("Failed to send onBattleStart callback",
			zap.Uint32("conn_id", ctx.ConnID),
			zap.Error(err))
	}

	return map[string]interface{}{
		"success":    true,
		"battleId":   battle.ID,
		"battleData": battleDTO,
	}, nil
}

					y := parts[4]

					label := fmt.Sprintf("%s - %s Ngân phiếu", name, cost)
					funcID := fmt.Sprintf("transport_%s_%s_%s_%s", mapID, cost, x, y)

					transportOptions = append(transportOptions, map[string]interface{}{
						"label": label,
						"func":  funcID,
					})
				}
			}

			if len(transportOptions) > 0 {
				npcName := npc.Name
				content := "Xin chào! Bạn muốn chuyển đến đâu?"
				if err := ctx.Connection.SendCallbackSync("onList", npcID, npcName, content, transportOptions); err != nil {
					h.logger.Error("Failed to send onList callback", zap.Error(err))
				}
				return map[string]interface{}{"success": true}, nil
			}
		}
	}

	// Determine NPC State and Quests
	npcState := 0
	questList := make(map[string]interface{})
	hasQuest := false

	if h.questService != nil {
		// 1. Get Available Quests (StartNPC)
		availableQuests, err := h.questService.GetAvailableQuestsForNpc(ctx.Context, characterID, npcID)
		if err == nil {
			for _, q := range availableQuests {
				hasQuest = true
				if npcState < 102 { // Priority: Complete (104) > Active (103) > Available (102)
					npcState = 102
				}
				questList[fmt.Sprintf("%d", q.ID)] = map[string]interface{}{
					"id":    q.ID,
					"qid":   q.ID,
					"name":  q.Name,
					"state": 102,
					"type":  q.Type,
				}
			}
		}

		// 2. Get Active/Completed Quests (FinishNPC or In-Progress)
		// We need to fetch ALL active quests and check if this NPC is involved
		activeQuests, err := h.questService.GetActiveQuests(ctx.Context, characterID)
		if err == nil {
			for _, qp := range activeQuests {
				if h.gameData != nil {
					qTpl := h.gameData.GetQuest(qp.QuestID)
					if qTpl != nil {
						// Check if this NPC is the FinishNPC
						if int(qTpl.FinishNPC) == npcID {
							hasQuest = true
							qState := 103 // In Progress
							if qp.IsComplete() {
								qState = 104 // Can Finish
							}

							// Update global NPC state priority
							if qState == 104 {
								npcState = 104
							} else if qState == 103 && npcState < 103 {
								npcState = 103
							}

							questList[fmt.Sprintf("%d", qp.QuestID)] = map[string]interface{}{
								"id":    qp.QuestID,
								"qid":   qp.QuestID,
								"name":  qTpl.Name,
								"state": qState,
								"type":  qTpl.Type,
							}
						}
					}
				}
			}
		}
	}

	if hasQuest {
		npcName := fmt.Sprintf("NPC %d", npcID)
		if h.gameData != nil {
			if npc := h.gameData.GetNPC(npcID); npc != nil {
				npcName = npc.Name
			}
		}

		npcData := map[string]interface{}{
			"npcId":   npcID,
			"npcName": npcName,
			"npcType": -1, // Default Normal
			"loopInfo": map[string]interface{}{
				"d":    nil,
				"flag": false,
			},
		}

		if directShopID > 0 {
			npcData["npcType"] = 2 // Shop NPC
		} else {
			npcData["npcType"] = -1 // Quest NPC
		}
		npcData["npcQuest"] = map[string]interface{}{
			"npcId": npcID,
			"state": npcState,
			"list":  questList,
		}

		if err := ctx.Connection.SendCallbackSync("npcFuncInit", npcData); err != nil {
			h.logger.Error("Failed to send npcFuncInit callback", zap.Error(err))
		}

		h.logger.Debug("NPC interaction initiated (Quest)",
			zap.Int64("character_id", characterID),
			zap.Int("npc_id", npcID),
			zap.Int("state", npcState))

		return map[string]interface{}{"success": true}, nil
	}
	// Have shop, No quest
	if directShopID > 0 {
		h.logger.Info("Directly opening shop for player",
			zap.Int64("character_id", characterID),
			zap.Int("npc_id", npcID),
			zap.Int("shop_id", directShopID))

		// Client ViewManager.as defines NPCSHOP_CONFIG only for IDs 1-7 (Special/Point Shops).
		// These must be used with "openShopById".
		// Standard shops (like ID 23) must be used with "onOpenShop".
		if directShopID <= 7 {
			if err := ctx.Connection.SendCallbackSync("openShopById", directShopID); err != nil {
				h.logger.Error("Failed to send openShopById callback", zap.Error(err))
			}
		} else {
			if err := ctx.Connection.SendCallbackSync("onOpenShop", directShopID); err != nil {
				h.logger.Error("Failed to send onOpenShop callback", zap.Error(err))
			}
		}
		return map[string]interface{}{"success": true}, nil
	}

	// Fallback for normal NPCs (No Quest, No Shop)
	npcName := fmt.Sprintf("NPC %d", npcID)
	npcType := -1

	if h.gameData != nil {
		if npc := h.gameData.GetNPC(npcID); npc != nil {
			npcName = npc.Name
			// Check for special types
			nType := int(npc.Type)
			if nType == 5 || nType == 7 || nType == 13 { // Type Bank
				npcType = nType
			}
		}
	}

	npcData := map[string]interface{}{
		"npcId":   npcID,
		"npcName": npcName,
		"npcType": npcType,
		"loopInfo": map[string]interface{}{
			"d":    nil,
			"flag": false,
		},
		"npcQuest": map[string]interface{}{
			"npcId": npcID,
			"state": -1,
			"list":  make(map[string]interface{}),
		},
	}

	if err := ctx.Connection.SendCallbackSync("npcFuncInit", npcData); err != nil {
		h.logger.Error("Failed to send npcFuncInit callback", zap.Error(err))
	}

	h.logger.Debug("NPC interaction initiated",
		zap.Int64("character_id", characterID),
		zap.Int("npc_id", npcID),
		zap.Int("state", npcState),
		zap.Bool("has_quest", hasQuest))

	return map[string]interface{}{"success": true}, nil
}

// CheckBattleOnMove checks for random battle encounters when player moves.
// Client call: remote.cbom(posX, posY)
// Parameters:
//   - posX (Number): Current X position
//   - posY (Number): Current Y position
//
// This is called periodically while player is moving to check for random encounters.
// If a battle should trigger, the server will start a battle and notify the client.
func (h *CombatHandler) CheckBattleOnMove(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	if len(args) < 2 {
		return nil, pkgerrors.ErrInvalidArgs
	}

	var posX, posY int
	switch v := args[0].(type) {
	case float64:
		posX = int(v)
	case int:
		posX = v
	default:
		return nil, pkgerrors.ErrInvalidArgs
	}

	switch v := args[1].(type) {
	case float64:
		posY = int(v)
	case int:
		posY = v
	default:
		return nil, pkgerrors.ErrInvalidArgs
	}

	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	char, err := h.charService.GetByID(ctx.Context, characterID)
	if err != nil {
		return nil, err
	}

	// Check if player is already in battle
	existingBattle, _ := h.combatService.GetPlayerBattle(characterID)
	if existingBattle != nil && existingBattle.IsActive() {
		// Already in battle, no need to check for new encounter
		return map[string]interface{}{
			"success":  true,
			"inBattle": true,
		}, nil
	}

	// Get current map ID
	currentMapID, _ := ctx.Connection.GetSceneInfo()
	if currentMapID == 0 {
		currentMapID = char.MapID
	}

	// Check if map is safe (no random battles)
	// If gameData is available, check map's safeFlag
	isSafeMap := false
	if h.gameData != nil {
		mapTemplate := h.gameData.GetMap(currentMapID)
		if mapTemplate != nil {
			// SafeFlag > 0 means safe map (no random battles)
			isSafeMap = mapTemplate.SafeFlag > 0
		}
	}

	if isSafeMap {
		// Safe map, no random battles
		return map[string]interface{}{
			"success":     true,
			"inBattle":    false,
			"encountered": false,
		}, nil
	}

	encounterRate := 0.10
	if h.gameData != nil {
		mapTemplate := h.gameData.GetMap(currentMapID)
		if mapTemplate != nil && mapTemplate.BasicBattleRate > 0 {
			encounterRate = mapTemplate.BasicBattleRate / 100.0
		}
	}

	roll := rand.Float64()

	h.logger.Debug("CheckBattleOnMove",
		zap.Int64("character_id", characterID),
		zap.Int("map_id", currentMapID),
		zap.Int("pos_x", posX),
		zap.Int("pos_y", posY),
		zap.Float64("encounter_rate", encounterRate),
		zap.Float64("roll", roll))

	if roll < encounterRate {
		// Random encounter triggered!
		if h.gameData != nil {
			mapCreatures := h.gameData.GetMapCreaturesByMapID(currentMapID)
			normalCreatures := make([]*models.MapCreatureTemplate, 0, len(mapCreatures))
			for _, mc := range mapCreatures {
				if mc.BossFlag == 0 {
					normalCreatures = append(normalCreatures, mc)
				}
			}

			if len(normalCreatures) > 0 {
				playerCount := 1
				enemyCount := h.getEnemyCountForPlayers(playerCount)

				if enemyCount > len(normalCreatures) {
					enemyCount = len(normalCreatures)
				}

				selectedMapCreatures := h.selectRandomMapCreatures(normalCreatures, enemyCount)
				creatureTemplates := make([]appcombat.NPCTemplateData, 0, len(selectedMapCreatures))
				for _, mc := range selectedMapCreatures {
					creature := h.gameData.GetCreature(int(mc.Cid))
					if creature == nil {
						continue
					}
					level := int(mc.Level)
					if level <= 0 {
						level = int(creature.UseLv)
					}
					if level <= 0 {
						level = 10
					}
					t := appcombat.CalculateCreatureStats(creature, level)
					t.Exp = int(mc.Exp)
					t.ExpMulti = mc.ExpMulti
					creatureTemplates = append(creatureTemplates, t)
				}

				if len(creatureTemplates) == 0 {
					h.logger.Warn("No valid creatures found for map", zap.Int("map_id", currentMapID))
					return map[string]interface{}{
						"success":     true,
						"inBattle":    false,
						"encountered": false,
					}, nil
				}

				h.logger.Info("Random encounter triggered",
					zap.Int64("character_id", characterID),
					zap.Int("map_id", currentMapID),
					zap.Int("player_count", playerCount),
					zap.Int("enemy_count", len(creatureTemplates)))

				battle, err := h.combatService.StartPVEBattleWithMultipleEnemies(ctx.Context, characterID, currentMapID, creatureTemplates)
				if err != nil {
					h.logger.Warn("Failed to start random encounter battle",
						zap.Int64("character_id", characterID),
						zap.Error(err))
					return map[string]interface{}{
						"success":     true,
						"inBattle":    false,
						"encountered": false,
					}, nil
				}

				if h.battleService != nil {
					h.battleService.SetTurnDeadline(battle.ID, fmt.Sprintf("%d", characterID))
				}

				ctx.Connection.SetBattleID(battle.ID)

				battleDTO := battle.ToDTO()
				cList := battleDTO["cList"]
				h.logger.Info("Battle cList data",
					zap.Any("cList", cList),
					zap.Bool("guest", false))

				firstNPCID := 0
				if len(creatureTemplates) > 0 {
					firstNPCID = creatureTemplates[0].ID
				}

				battleData := map[string]interface{}{
					"battleFieldId": battle.ID,
					"nid":           firstNPCID,
					"battleType":    int(combat.BattleTypePVE),
					"mapId":         currentMapID,
					"battleData":    battleDTO,
					"cList":         cList,
					"guest":         false,
					"sneakFlag":     0,
				}

				if err := ctx.Connection.SendCallback("onBattleStart", battleData); err != nil {
					h.logger.Error("Failed to send onBattleStart callback",
						zap.Uint32("conn_id", ctx.ConnID),
						zap.Error(err))
				}

				return map[string]interface{}{
					"success":     true,
					"inBattle":    true,
					"encountered": true,
					"battleId":    battle.ID,
					"npcId":       firstNPCID,
				}, nil
			}
		}
	}

	// No encounter this time
	return map[string]interface{}{
		"success":     true,
		"inBattle":    false,
		"encountered": false,
	}, nil
}

func (h *CombatHandler) CombatAction(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	if len(args) < 4 {
		return nil, pkgerrors.ErrInvalidArgs
	}

	var battleID string
	var actionType, skillID int
	var targetID string

	switch v := args[0].(type) {
	case string:
		battleID = v
	default:
		return nil, pkgerrors.ErrInvalidArgs
	}

	switch v := args[1].(type) {
	case float64:
		actionType = int(v)
	case int:
		actionType = v
	default:
		return nil, pkgerrors.ErrInvalidArgs
	}

	switch v := args[2].(type) {
	case float64:
		skillID = int(v)
	case int:
		skillID = v
	default:
		return nil, pkgerrors.ErrInvalidArgs
	}

	switch v := args[3].(type) {
	case string:
		targetID = v
	case float64:
		targetID = string(rune(int(v)))
	case int:
		targetID = string(rune(v))
	default:
		targetID = ""
	}

	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	playerAction, err := h.combatService.ExecuteAction(
		ctx.Context,
		battleID,
		fmt.Sprintf("%d", characterID),
		combat.ActionType(actionType),
		skillID,
		targetID,
	)
	if err != nil {
		return rtmp.ErrorToResponse(err), nil
	}

	battle, err := h.combatService.GetBattle(battleID)
	if err != nil {
		return rtmp.ErrorToResponse(err), nil
	}

	var npcActions []*combat.BattleAction
	if battle.IsActive() {
		npcActions = h.combatService.ProcessNPCTurn(ctx.Context, battle)
	}

	actionDTOs := make([]map[string]interface{}, 0)
	actionDTOs = append(actionDTOs, playerAction.ToDTO())
	for _, a := range npcActions {
		actionDTOs = append(actionDTOs, a.ToDTO())
	}

	response := map[string]interface{}{
		"success":      true,
		"battleId":     battleID,
		"round":        battle.CurrentRound,
		"actions":      actionDTOs,
		"participants": nil,
	}

	participants := make([]map[string]interface{}, len(battle.Participants))
	for i, p := range battle.Participants {
		participants[i] = p.ToDTO()
	}
	response["participants"] = participants

	if battle.IsEnded() && battle.Result != nil {
		response["battleEnded"] = true
		response["result"] = battle.Result.ToDTO()
	}

	h.logger.Debug("Combat action processed",
		zap.String("battle_id", battleID),
		zap.Int64("character_id", characterID),
		zap.Int("skill_id", skillID))

	return response, nil
}

func (h *CombatHandler) EndBattle(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	var battleID string
	if len(args) >= 1 && args[0] != nil {
		switch v := args[0].(type) {
		case string:
			battleID = v
		}
	}
	if battleID == "" {
		battleID = ctx.Connection.GetBattleID()
	}
	if battleID == "" {
		return map[string]interface{}{
			"success": true,
			"rewards": map[string]interface{}{
				"exp":   0,
				"money": 0,
				"items": []interface{}{},
			},
		}, nil
	}

	battle, err := h.combatService.GetBattle(battleID)
	if err != nil {
		return map[string]interface{}{
			"success": true,
			"rewards": map[string]interface{}{
				"exp":   0,
				"money": 0,
				"items": []interface{}{},
			},
		}, nil
	}

	if battle.IsActive() {
		ended, winner := battle.CheckVictory()
		if !ended {
			winner = combat.SideEnemy
		}
		h.combatService.EndBattle(ctx.Context, battle, winner)

		// Handle quest updates if player won
		if winner == combat.SidePlayer && h.questService != nil {
			characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
			if err == nil {
				enemies := battle.GetParticipantsBySide(combat.SideEnemy)
				for _, enemy := range enemies {
					if enemy.EntityType == combat.ParticipantTypeCreature {
						updatedQuests, err := h.questService.OnMonsterKilled(ctx.Context, characterID, enemy.EntityID)
						if err != nil {
							h.logger.Warn("Failed to update quest progress", zap.Error(err))
							continue
						}
						// Send callbacks for updated quests
						for _, qp := range updatedQuests {
							dto := h.questService.EnrichQuestDTO(ctx.Context, qp)
							if err := ctx.Connection.SendCallback("onAddChaQuest", dto); err != nil {
								h.logger.Warn("Failed to send onAddChaQuest callback",
									zap.Int64("char_id", characterID),
									zap.Int("quest_id", qp.QuestID),
									zap.Error(err))
							}

							// If quest is complete, push NPC state update
							if qp.IsComplete() && h.questNotify != nil {
								questTemplate := h.gameData.GetQuest(qp.QuestID)
								if questTemplate != nil {
									finishNPC := int(questTemplate.FinishNPC)
									if finishNPC > 0 {
										h.questNotify.PushNpcQuestStates(ctx, characterID, []int{finishNPC})
									}
								}
							}
						}
					}
				}
			}
		}
	}

	characterID, _ := strconv.ParseInt(ctx.CharacterID, 10, 64)
	playerParticipant := battle.GetParticipant(fmt.Sprintf("%d", characterID))
	if playerParticipant != nil {
		char, charErr := h.charService.GetByID(ctx.Context, characterID)
		if charErr == nil {
			char.CurrentHP = playerParticipant.CurrentHP
			char.CurrentMP = playerParticipant.CurrentMP
			if err := h.charService.Save(ctx.Context, char); err != nil {
				h.logger.Warn("Failed to save character HP after battle", zap.Error(err))
			}
			ctx.Connection.SendCallback("onUPP", map[string]interface{}{
				"currentHp": char.CurrentHP,
				"currentMp": char.CurrentMP,
			})
		}
	}

	if h.battleService != nil {
		h.battleService.ClearBattleDeadlines(battleID)
	}

	ctx.Connection.ClearBattle()

	result := battle.Result
	if result == nil {
		result = &combat.BattleResult{
			WinnerSide:  combat.SideEnemy,
			ItemRewards: []int{},
		}
	}

	h.logger.Info("Battle play ended",
		zap.String("battle_id", battleID),
		zap.Int64("exp_reward", result.ExpReward),
		zap.Int64("money_reward", result.MoneyReward))

	return map[string]interface{}{
		"success": true,
		"rewards": result.ToDTO(),
	}, nil
}

func (h *CombatHandler) BattleUpdateCmd(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	if len(args) < 1 {
		h.logger.Warn("BattleUpdateCmd called with no arguments")
		return nil, pkgerrors.ErrInvalidArgs
	}

	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	battle, err := h.combatService.GetPlayerBattle(characterID)
	if err != nil {
		return rtmp.ErrorToResponse(err), nil
	}

	if h.battleService != nil {
		h.battleService.ClearTurnDeadline(battle.ID, fmt.Sprintf("%d", characterID))
	}

	if !battle.IsActive() {
		return map[string]interface{}{
			"success": false,
			"error":   "Battle is not active",
		}, nil
	}

	var cmdArray []interface{}
	switch v := args[0].(type) {
	case []interface{}:
		cmdArray = v
	case amf0.ECMAArray:
		maxIdx := 0
		for key := range v {
			if idx, err := strconv.Atoi(key); err == nil && idx >= maxIdx {
				maxIdx = idx + 1
			}
		}
		cmdArray = make([]interface{}, maxIdx)
		for key, val := range v {
			if idx, err := strconv.Atoi(key); err == nil && idx >= 0 && idx < len(cmdArray) {
				cmdArray[idx] = val
			}
		}
		h.logger.Debug("BattleUpdateCmd parsed ECMAArray", zap.Int("count", len(cmdArray)))
	case map[string]interface{}:
		maxIdx := 0
		for key := range v {
			if idx, err := strconv.Atoi(key); err == nil && idx >= maxIdx {
				maxIdx = idx + 1
			}
		}
		cmdArray = make([]interface{}, maxIdx)
		for key, val := range v {
			if idx, err := strconv.Atoi(key); err == nil && idx >= 0 && idx < len(cmdArray) {
				cmdArray[idx] = val
			}
		}
	case nil:
		h.logger.Warn("BattleUpdateCmd received nil argument, treating as empty array")
		cmdArray = []interface{}{}
	default:
		h.logger.Warn("BattleUpdateCmd received unexpected argument type",
			zap.String("type", fmt.Sprintf("%T", args[0])),
			zap.Any("value", args[0]))
		return nil, pkgerrors.ErrInvalidArgs
	}

	commands := h.parseCommands(cmdArray, characterID)

	participantIDs := make([]string, len(battle.Participants))
	for i, p := range battle.Participants {
		participantIDs[i] = p.ID
	}

	hasNewCommands := len(commands) > 0
	if hasNewCommands {
		h.logger.Info("BattleUpdateCmd participants",
			zap.String("battle_id", battle.ID),
			zap.Strings("participant_ids", participantIDs),
			zap.Int("command_count", len(commands)))

		cmdMap := make(map[string]*combat.BattleCommand)
		for _, cmd := range commands {
			cmdMap[cmd.ActorID] = cmd
		}
		battle.SetPendingCommands(cmdMap)

		for _, cmd := range commands {
			h.logger.Info("BattleUpdateCmd command",
				zap.String("actor_id", cmd.ActorID),
				zap.String("target_id", cmd.TargetID),
				zap.Int("action_type", cmd.ActionType))
		}
	} else {
		h.logger.Info("BattleUpdateCmd continuation call",
			zap.String("battle_id", battle.ID),
			zap.Bool("has_pending_commands", battle.HasPendingCommands()))
	}

	if !battle.HasPendingCommands() {
		h.logger.Warn("No pending commands to process")
		return map[string]interface{}{
			"success": false,
			"error":   "No pending commands",
		}, nil
	}

	processor := combat.NewTurnProcessor(battle, nil)
	pendingCmds := battle.GetPendingCommands()
	for _, cmd := range pendingCmds {
		processor.SetCommand(cmd)
	}

	result := processor.ProcessRound()

	actionDTOs := make([]map[string]interface{}, 0, len(result.Actions))
	skipActionDTOs := make([]map[string]interface{}, 0)
	for i, a := range result.Actions {
		dto := a.ToDTO()
		if effects, ok := dto["effects"].([]string); ok && len(effects) > 0 && effects[0] == "skip" {
			skipActionDTOs = append(skipActionDTOs, dto)
			h.logger.Info("Skip action for dead actor", zap.Int("index", i), zap.Any("sid", dto["sid"]))
		} else {
			actionDTOs = append(actionDTOs, dto)
		}
	}

	if result.BattleEnded {
		winType := 0
		if result.Winner == combat.SidePlayer {
			winType = 1
		}
		endAction := map[string]interface{}{
			"end":    true,
			"type":   winType,
			"result": 1,
		}
		actionDTOs = append(actionDTOs, endAction)
		h.logger.Info("Battle ended, added end marker", zap.Int("winner", int(result.Winner)))
		battle.SetRoundComplete()

		playerParticipant := battle.GetParticipant(fmt.Sprintf("%d", characterID))
		if playerParticipant != nil {
			char, charErr := h.charService.GetByID(ctx.Context, characterID)
			if charErr == nil {
				char.CurrentHP = playerParticipant.CurrentHP
				char.CurrentMP = playerParticipant.CurrentMP
				if err := h.charService.Save(ctx.Context, char); err != nil {
					h.logger.Warn("Failed to save character HP after battle in BattleUpdateCmd", zap.Error(err))
				}
			}
			ctx.Connection.SendCallback("onUPP", map[string]interface{}{
				"currentHp": playerParticipant.CurrentHP,
				"currentMp": playerParticipant.CurrentMP,
			})
		}
	}

	cList := make([]map[string]interface{}, len(battle.Participants))
	for i, p := range battle.Participants {
		cList[i] = p.ToDTO()
	}

	response := map[string]interface{}{
		"success":  true,
		"battleId": battle.ID,
		"round":    result.Round,
		"actions":  actionDTOs,
		"cList":    cList,
	}

	if result.BattleEnded {
		response["battleEnded"] = true
	}

	if result.MoreTurns && !result.BattleEnded {
		response["moreTurns"] = true
		h.logger.Info("More turns remaining in round", zap.Int("round", result.Round))
	}

	for i, actionDTO := range actionDTOs {
		h.logger.Info("Action DTO",
			zap.Int("index", i),
			zap.Any("sid", actionDTO["sid"]),
			zap.Any("tid", actionDTO["tid"]),
			zap.Any("bid", actionDTO["bid"]),
			zap.Any("sSObj", actionDTO["sSObj"]),
			zap.Any("tSObj", actionDTO["tSObj"]),
			zap.Any("delay", actionDTO["delay"]),
			zap.Any("end", actionDTO["end"]))
	}

	nestedActions := make([]interface{}, 0, len(actionDTOs))
	for i := 0; i < len(actionDTOs); i++ {
		dto := actionDTOs[i]
		bid, _ := dto["bid"]
		bidVal, _ := bid.(int)
		if i+1 < len(actionDTOs) {
			nextBid, _ := actionDTOs[i+1]["bid"]
			nextBidVal, _ := nextBid.(int)
			if (bidVal == 3 || bidVal == 7) && nextBidVal == 5 {
				nestedActions = append(nestedActions, []map[string]interface{}{dto, actionDTOs[i+1]})
				i++
				continue
			}
		}
		nestedActions = append(nestedActions, []map[string]interface{}{dto})
	}
	h.logger.Info("Action sequence created", zap.Int("total_actions", len(nestedActions)))

	if err := ctx.Connection.SendCallback("onBattlePlayList", nestedActions); err != nil {
		h.logger.Error("Failed to send onBattlePlayList callback", zap.Error(err))
	}
	h.logger.Info("Sent onBattlePlayList callback", zap.Int("steps", len(actionDTOs)))

	if !result.BattleEnded && !result.MoreTurns {
		if h.battleService != nil {
			h.battleService.SetTurnDeadline(battle.ID, fmt.Sprintf("%d", characterID))
			h.battleService.OnRoundProcessed()
		}
	}

	h.logger.Debug("Battle round processed",
		zap.String("battle_id", battle.ID),
		zap.Int64("character_id", characterID),
		zap.Int("round", result.Round),
		zap.Int("actions", len(result.Actions)),
		zap.Bool("more_turns", result.MoreTurns))

	return response, nil
}

func (h *CombatHandler) parseCommands(cmdArray []interface{}, characterID int64) []*combat.BattleCommand {
	commands := make([]*combat.BattleCommand, 0, len(cmdArray))

	for i, item := range cmdArray {
		var cmdMap map[string]interface{}
		switch v := item.(type) {
		case map[string]interface{}:
			cmdMap = v
		case amf0.ECMAArray:
			cmdMap = map[string]interface{}(v)
		default:
			h.logger.Debug("parseCommands skipping non-map item", zap.Int("index", i), zap.String("type", fmt.Sprintf("%T", item)))
			continue
		}

		cmd := &combat.BattleCommand{}

		if i == 0 {
			cmd.ActorID = fmt.Sprintf("%d", characterID)
		} else {
			cmd.ActorID = fmt.Sprintf("%d_pet", characterID)
		}

		if tid, ok := cmdMap["tid"]; ok {
			cmd.TargetID = h.parseTargetID(tid)
		}

		if action, ok := cmdMap["action"]; ok {
			switch v := action.(type) {
			case float64:
				cmd.ActionType = int(v)
			case int:
				cmd.ActionType = v
			}
		}

		if id, ok := cmdMap["id"]; ok {
			switch v := id.(type) {
			case float64:
				cmd.SkillID = int(v)
				cmd.ItemSlot = int(v)
			case int:
				cmd.SkillID = v
				cmd.ItemSlot = v
			}
		}

		if level, ok := cmdMap["level"]; ok {
			switch v := level.(type) {
			case float64:
				cmd.SkillLevel = int(v)
			case int:
				cmd.SkillLevel = v
			}
		}

		h.logger.Debug("Parsed battle command",
			zap.String("actor_id", cmd.ActorID),
			zap.String("target_id", cmd.TargetID),
			zap.Int("action_type", cmd.ActionType),
			zap.Int("skill_id", cmd.SkillID),
			zap.Int("skill_level", cmd.SkillLevel))

		commands = append(commands, cmd)
	}

	return commands
}

func (h *CombatHandler) parseTargetID(tid interface{}) string {
	switch v := tid.(type) {
	case string:
		return v
	case float64:
		return fmt.Sprintf("%d", int(v))
	case int:
		return fmt.Sprintf("%d", v)
	default:
		return ""
	}
}

func (h *CombatHandler) getEnemyCountForPlayers(playerCount int) int {
	var minEnemies, maxEnemies int
	switch playerCount {
	case 1:
		minEnemies, maxEnemies = 1, 2
	case 2:
		minEnemies, maxEnemies = 2, 4
	case 3:
		minEnemies, maxEnemies = 3, 5
	case 4, 5:
		minEnemies, maxEnemies = 4, 5
	default:
		minEnemies, maxEnemies = 1, 2
	}
	return minEnemies + rand.Intn(maxEnemies-minEnemies+1)
}

func (h *CombatHandler) selectRandomMapCreatures(entries []*models.MapCreatureTemplate, count int) []*models.MapCreatureTemplate {
	if count >= len(entries) {
		return entries
	}

	shuffled := make([]*models.MapCreatureTemplate, len(entries))
	copy(shuffled, entries)
	rand.Shuffle(len(shuffled), func(i, j int) {
		shuffled[i], shuffled[j] = shuffled[j], shuffled[i]
	})

	return shuffled[:count]
}

func (h *CombatHandler) MazeRecover(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	if len(args) < 1 {
		return nil, pkgerrors.ErrInvalidArgs
	}

	var hp int
	switch v := args[0].(type) {
	case float64:
		hp = int(v)
	case int:
		hp = v
	default:
		return nil, pkgerrors.ErrInvalidArgs
	}

	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	h.logger.Info("Maze recover",
		zap.Int64("character_id", characterID),
		zap.Int("hp", hp))

	char, err := h.charService.GetByID(ctx.Context, characterID)
	if err != nil {
		return rtmp.ErrorToResponse(err), nil
	}

	if hp > 0 {
		char.CurrentHP += hp
		if char.CurrentHP > char.MaxHP {
			char.CurrentHP = char.MaxHP
		}

		ctx.Connection.SendCallback("onUPP", map[string]interface{}{
			"hp": char.CurrentHP,
		})
	}

	return map[string]interface{}{
		"success": true,
	}, nil
}

func (h *CombatHandler) teleportToMap(ctx *rtmp.RPCContext, characterID int64, mapID int) error {
	safeX, safeY := h.sceneService.GetSafePoint(mapID)

	char, err := h.charService.GetByID(ctx.Context, characterID)
	if err != nil {
		return err
	}

	oldRoomID, _ := ctx.Connection.GetSceneInfo()
	channelID := ctx.Connection.GetChannelID()

	if oldRoomID != 0 && oldRoomID != mapID {
		h.sceneManager.BroadcastToScene(channelID, oldRoomID, ctx.ConnID, "onCharLeaveScene", characterID)
		ctx.Connection.SetSceneInfo(mapID, characterID)
		h.sceneManager.MoveToScene(channelID, oldRoomID, mapID, ctx.Connection)
	} else if oldRoomID == 0 {
		ctx.Connection.SetSceneInfo(mapID, characterID)
		h.sceneManager.AddToScene(channelID, mapID, ctx.Connection)
	}

	if err := h.charService.UpdatePosition(ctx.Context, characterID, character.Position{MapID: mapID, X: safeX, Y: safeY}); err != nil {
		h.logger.Error("Failed to update position", zap.Error(err))
	}
	ctx.Connection.SendCallback("onSceneEnter", mapID, safeX, safeY, -1)

	if oldRoomID != 0 && oldRoomID != mapID {
		char.MapID = mapID
		char.PosX = safeX
		char.PosY = safeY
		playerData := h.sceneService.GetCharacterForClient(char)
		h.sceneManager.BroadcastToScene(channelID, mapID, ctx.ConnID, "onScenePlayerEntered", playerData)
	}
	return nil
}
