type TradeHandler struct {
	rtmpServer *rtmp.Server
	logger     *zap.Logger
}

func NewTradeHandler(rtmpServer *rtmp.Server, logger *zap.Logger) *TradeHandler {
	return &TradeHandler{
		rtmpServer: rtmpServer,
		logger:     logger,
	}
}

func (h *TradeHandler) RegisterHandlers(dispatcher *rtmp.RPCDispatcher) {
	dispatcher.Register("newTrade", h.NewTrade)
	dispatcher.Register("stopTrade", h.StopTrade)
	dispatcher.Register("tradeLock", h.TradeLock)
	dispatcher.Register("tradeConfirm", h.TradeConfirm)
	dispatcher.Register("tradeAddItem", h.TradeAddItem)
	dispatcher.Register("tradeRemoveItem", h.TradeRemoveItem)
	dispatcher.Register("tradeSetMoney", h.TradeSetMoney)
}

// NewTrade initiates a new trade request with another player.
// Client call: _core.remote.call("newTrade", responder, targetId)
// Parameters: targetId (Number) - target player's character ID
// Response: {flag: true/false, info: string} or {tradeId: number}
func (h *TradeHandler) NewTrade(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	h.logger.Info("NewTrade called",
		zap.Uint32("conn_id", ctx.ConnID),
		zap.String("character_id", ctx.CharacterID),
		zap.Any("args", args))

	if len(args) < 1 {
		return map[string]interface{}{
			"flag": false,
			"info": "Missing target player ID",
		}, nil
	}

	targetIDFloat, ok := args[0].(float64)
	if !ok {
		return map[string]interface{}{
			"flag": false,
			"info": "Invalid target player ID",
		}, nil
	}

	targetID := int64(targetIDFloat)
	initiatorID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		h.logger.Warn("NewTrade: invalid character ID",
			zap.String("character_id", ctx.CharacterID),
			zap.Error(err))
		return map[string]interface{}{
			"flag": false,
			"info": "Invalid character ID",
		}, nil
	}

	// Check if trying to trade with self
	if targetID == initiatorID {
		return map[string]interface{}{
			"flag": false,
			"info": "Cannot trade with yourself",
		}, nil
	}

	// Check if target player is online
	targetIDStr := strconv.FormatInt(targetID, 10)
	targetConn := h.rtmpServer.GetConnectionByCharacterID(targetIDStr)
	if targetConn == nil {
		return map[string]interface{}{
			"flag": false,
			"info": "Target player is not online",
		}, nil
	}

	// TODO: Check if target player is already in a trade
	// TODO: Check distance between players
	// TODO: Create trade session and store in memory/database
	// TODO: Send trade request notification to target player
	// TODO: Generate trade ID

	// For now, return success to allow client to proceed
	// In real implementation, we would:
	// 1. Create a trade session
	// 2. Send notification to target player
	// 3. Wait for target player to accept
	// 4. Return trade ID when both players are ready

	h.logger.Debug("Trade request initiated",
		zap.Int64("initiator_id", initiatorID),
		zap.Int64("target_id", targetID))

	// Mock: Return success with a mock trade ID
	// Client expects either {flag: true} or {flag: false, info: string}
	return map[string]interface{}{
		"flag":    true,
		"tradeId": initiatorID*10000 + targetID, // Mock trade ID
	}, nil
}
