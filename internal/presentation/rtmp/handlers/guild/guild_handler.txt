// InitViewGuildP initializes the guild panel view.
// Client call: _core.remote.initViewGuildP()
// Parameters: None
//
// Response: Sends callback onInitViewGuildP({myGuild, memberList, guildRank, applyList, skillDevData})
//   - myGuild: Object containing guild information:
//   - name: Guild name
//   - ln: Leader name
//   - guildInfo: Guild description/info
//   - money: Guild money
//   - exp: Guild experience
//   - bagSlotNum: Number of bag slots
//   - memberList: Array of member objects, each with:
//   - cid: Character ID
//   - rank: Member rank
//   - name: Character name
//   - level: Character level
//   - class: Character class
//   - online: Online status
//   - contribution: Contribution points
//   - guildRank: Array of rank objects, each with:
//   - rank: Rank ID
//   - name: Rank name
//   - permissions: Rank permissions
//   - applyList: Array of application objects
//   - skillDevData: Object containing guild skill development data
func (h *GuildHandler) InitViewGuildP(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	charIDStr := ctx.CharacterID
	if charIDStr == "" {
		h.logger.Warn("InitViewGuildP: no character ID in session",
			zap.Uint32("conn_id", ctx.ConnID))
		return nil, pkgerrors.ErrUnauthorized
	}

	characterID, err := strconv.ParseInt(charIDStr, 10, 64)
	if err != nil {
		h.logger.Warn("InitViewGuildP: invalid character ID",
			zap.String("char_id", charIDStr),
			zap.Error(err))
		return nil, pkgerrors.ErrInvalidInput
	}

	h.logger.Debug("InitViewGuildP: initializing guild panel",
		zap.Uint32("conn_id", ctx.ConnID),
		zap.Int64("character_id", characterID))

	// Default response - no guild (empty data)
	response := map[string]interface{}{
		"myGuild":      nil,
		"memberList":   []interface{}{},
		"guildRank":    []interface{}{},
		"applyList":    []interface{}{},
		"skillDevData": nil,
	}

	// TODO: In a full implementation:
	// - Check if character has a guild
	// - If yes, fetch guild data:
	//   - Guild info (name, leader, description, money, exp, etc.)
	//   - Member list with their ranks and status
	//   - Guild rank definitions
	//   - Pending applications
	//   - Skill development progress
	// - If no guild, return empty/null data
	// Fetch guild data if service is available
	if h.guildService != nil {
		g, err := h.guildService.GetGuildByMember(context.Background(), characterID)
		if err != nil {
			h.logger.Warn("InitViewGuildP: failed to get guild", zap.Error(err))
		} else if g != nil {
			// Found guild, populate response

			// Find leader name
			leaderName := ""
			memberList := make([]interface{}, 0, len(g.Members))

			for _, m := range g.Members {
				if m.Rank == 0 { // Leader
					leaderName = m.CharacterName
				}
				memberList = append(memberList, m.ToDTO())
			}

			// Convert guild to DTO
			guildDTO := g.ToDTO()
			guildDTO["ln"] = leaderName // Override ID with Name as expected by client
			guildDTO["bagSlotNum"] = 28 // Default guild bank slots

			response["myGuild"] = guildDTO
			response["memberList"] = memberList

			// Define default guild ranks (0=Leader, 1=Vice, etc)
			// This typically matches client static data but server can override
			response["guildRank"] = []interface{}{
				map[string]interface{}{"rank": 0, "name": "Bang cheng", "canAdd": 1, "canQuest": 1, "canSlot": 1, "canInfo": 1, "canDel": 1, "canDuty": 1},
				map[string]interface{}{"rank": 1, "name": "Pho bang", "canAdd": 1, "canQuest": 1, "canSlot": 1, "canInfo": 1, "canDel": 1, "canDuty": 1},
				map[string]interface{}{"rank": 2, "name": "Tinh anh", "canAdd": 1, "canQuest": 1, "canSlot": 0, "canInfo": 0, "canDel": 0, "canDuty": 0},
				map[string]interface{}{"rank": 3, "name": "Thanh vien", "canAdd": 0, "canQuest": 0, "canSlot": 0, "canInfo": 0, "canDel": 0, "canDuty": 0},
			}
		}
	} else {
		// Mock data for development if service not injected (fallback)
		// ... or just leave empty
	}

	h.logger.Info("InitViewGuildP: returning guild panel data",
		zap.Uint32("conn_id", ctx.ConnID),
		zap.Int64("character_id", characterID),
		zap.Bool("has_guild", response["myGuild"] != nil))

	// Send callback
	if err := ctx.Connection.SendCallback("onInitViewGuildP", response); err != nil {
		h.logger.Error("InitViewGuildP: failed to send callback",
			zap.Uint32("conn_id", ctx.ConnID),
			zap.Error(err))
	}

	return response, nil
}

// AddGuild creates a new guild.
// Client call: _core.remote.addGuild({name: guildName})
// Parameters:
//   - name (String): Guild name
//
// Response: Sends callback onAddGuild({id, name, leader, level, etc.})
func (h *GuildHandler) AddGuild(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	charIDStr := ctx.CharacterID
	if charIDStr == "" {
		h.logger.Warn("AddGuild: no character ID in session",
			zap.Uint32("conn_id", ctx.ConnID))
		return nil, pkgerrors.ErrUnauthorized
	}

	characterID, err := strconv.ParseInt(charIDStr, 10, 64)
	if err != nil {
		h.logger.Warn("AddGuild: invalid character ID",
			zap.String("char_id", charIDStr),
			zap.Error(err))
		return nil, pkgerrors.ErrInvalidInput
	}

	if len(args) < 1 {
		h.logger.Warn("AddGuild: missing guild data",
			zap.Uint32("conn_id", ctx.ConnID))
		return nil, pkgerrors.ErrInvalidArgs
	}

	guildData, ok := args[0].(map[string]interface{})
	if !ok {
		h.logger.Warn("AddGuild: invalid guild data format",
			zap.Any("data", args[0]))
		return nil, pkgerrors.ErrInvalidInput
	}

	guildName, _ := guildData["name"].(string)
	if guildName == "" {
		h.logger.Warn("AddGuild: missing guild name")
		return nil, pkgerrors.ErrInvalidInput
	}

	h.logger.Info("AddGuild: creating new guild",
		zap.Uint32("conn_id", ctx.ConnID),
		zap.Int64("character_id", characterID),
		zap.String("guild_name", guildName))

	// Fetch character name for leader
	var leaderName string
	var guildID int64 = 1 // Default ID if service not available

	if h.charService != nil {
		char, err := h.charService.GetByID(context.Background(), characterID)
		if err != nil {
			h.logger.Error("AddGuild: failed to get character", zap.Error(err))
			return nil, pkgerrors.ErrSystemError
		}
		if char == nil {
			h.logger.Warn("AddGuild: character not found", zap.Int64("character_id", characterID))
			return nil, pkgerrors.ErrCharacterNotFound
		}

		leaderName = char.Name

		// Check level requirement (Level 50)
		if char.Level < 50 {
			h.logger.Warn("AddGuild: insufficient level",
				zap.Int64("character_id", characterID),
				zap.Int("level", char.Level),
				zap.Int("required", 50))
			return nil, pkgerrors.ErrInsufficientLevel
		}

		// Check money requirement (1,000,000 MoneyBind)
		const requiredMoney = 1000000
		if char.MoneyBind < requiredMoney {
			h.logger.Warn("AddGuild: insufficient bound money",
				zap.Int64("character_id", characterID),
				zap.Int64("money_bind", char.MoneyBind),
				zap.Int64("required", requiredMoney))
			return nil, pkgerrors.ErrInsufficientFunds
		}

		// Create guild first
		if h.guildService != nil {
			g, err := h.guildService.CreateGuild(context.Background(), characterID, guildName)
			if err != nil {
				h.logger.Warn("AddGuild: failed to create guild", zap.Error(err))
				return nil, err // Pass through service errors (name taken, etc)
			}
			guildID = g.ID
		}

		// Deduct money
		char.MoneyBind -= requiredMoney
		if err := h.charService.Update(context.Background(), char); err != nil {
			// Critical error: Guild created but money not deducted
			// In a real system, we would need to roll back the guild creation
			h.logger.Error("AddGuild: failed to update character money", zap.Error(err))
			// Attempt to rollback guild creation
			if h.guildService != nil {
				_ = h.guildService.DisbandGuild(context.Background(), guildID, characterID)
			}
			return nil, pkgerrors.ErrSystemError
		}

		_ = ctx.Connection.SendCallback("onUPP", map[string]interface{}{
			"moneyBind": char.MoneyBind,
		})

		h.logger.Info("AddGuild: deducted creation fee",
			zap.Int64("character_id", characterID),
			zap.Int64("amount", requiredMoney),
			zap.Int64("remaining", char.MoneyBind))
	}

	// Response
	response := map[string]interface{}{
		"id":          guildID,
		"name":        guildName,
		"cid":         characterID, // Character ID of the leader
		"ln":          leaderName,  // Leader name
		"level":       1,
		"memberCount": 1,
	}

	h.logger.Info("AddGuild: guild created",
		zap.Uint32("conn_id", ctx.ConnID),
		zap.String("guild_name", guildName),
		zap.String("leader_name", leaderName))

	// Send callback
	if err := ctx.Connection.SendCallback("onAddGuild", response); err != nil {
		h.logger.Error("AddGuild: failed to send callback",
			zap.Uint32("conn_id", ctx.ConnID),
			zap.Error(err))
	}

	// Automatically refresh guild panel for the client
	_, _ = h.InitViewGuildP(ctx, nil)

	return response, nil
}
