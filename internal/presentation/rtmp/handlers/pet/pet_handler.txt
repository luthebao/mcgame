func (h *PetHandler) InitViewPetMngP(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	pets, err := h.petService.GetPetList(ctx.Context, characterID)
	if err != nil {
		return rtmp.ErrorToResponse(err), nil
	}

	petDTOs := make([]map[string]interface{}, len(pets))
	for i, p := range pets {
		// ToDTO wraps the pet data in a "data" key, which is used for single pet updates.
		// For the list view, the client expects the pet objects directly in the array.
		if data, ok := p.ToDTO()["data"].(map[string]interface{}); ok {
			petDTOs[i] = data
		} else {
			petDTOs[i] = p.ToDTO()
		}
	}

	h.logger.Debug("Pet management panel initialized",
		zap.Int64("character_id", characterID),
		zap.Int("pet_count", len(pets)))

	return petDTOs, nil
}

func (h *PetHandler) PetOpenSlot(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	newSlotCount, cost, currencyType, remaining, err := h.petService.OpenPetSlot(ctx.Context, characterID)
	if err != nil {
		h.logger.Error("Failed to open pet slot", zap.Error(err))
		return rtmp.ErrorToResponse(err), nil
	}

	// Callback: onMinusMoney 11581 goldBind 20 4510
	// 11581 is likely charID.
	_ = ctx.Connection.SendCallback("onMinusMoney", float64(characterID), currencyType, float64(cost), float64(remaining))

	// Callback: onUPP petMaxNum: 6
	_ = ctx.Connection.SendCallback("onUPP", map[string]interface{}{
		"petMaxNum": newSlotCount,
	})

	msg := fmt.Sprintf("Mất %d kim phiếu", cost)
	_ = ctx.Connection.SendCallback("onSystemSay", msg)

	h.logger.Info("Pet slot opened",
		zap.Int64("character_id", characterID),
		zap.Int("new_slots", newSlotCount))

	return map[string]interface{}{
		"success": true,
		"ps":      newSlotCount,
	}, nil
}

// ChangePetState handles client's changePetState(petID, state) call.
// State 1 = following (ra trận), State 3 = rest (trong kho)
func (h *PetHandler) ChangePetState(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	if len(args) < 2 {
		return nil, pkgerrors.ErrInvalidArgs
	}

	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	var petID int64
	switch v := args[0].(type) {
	case float64:
		petID = int64(v)
	case int64:
		petID = v
	case int:
		petID = int64(v)
	default:
		return nil, pkgerrors.ErrInvalidArgs
	}

	var newState int
	switch v := args[1].(type) {
	case float64:
		newState = int(v)
	case int:
		newState = v
	default:
		return nil, pkgerrors.ErrInvalidArgs
	}

	h.logger.Info("ChangePetState called",
		zap.Int64("character_id", characterID),
		zap.Int64("pet_id", petID),
		zap.Int("new_state", newState))

	var p *pet.Pet

	switch newState {
	case 1: // Following (ra trận)
		p, err = h.petService.StartFollow(ctx.Context, characterID, petID, 0)
		if err != nil {
			h.logger.Error("Failed to start pet follow", zap.Error(err))
			return rtmp.ErrorToResponse(err), nil
		}
	case 3: // Rest (trong kho)
		err = h.petService.CancelFollow(ctx.Context, characterID, petID)
		if err != nil {
			h.logger.Error("Failed to cancel pet follow", zap.Error(err))
			return rtmp.ErrorToResponse(err), nil
		}
		// Get updated pet for DTO
		p, err = h.petService.GetPetDetail(ctx.Context, characterID, petID)
		if err != nil {
			return rtmp.ErrorToResponse(err), nil
		}
	default:
		h.logger.Warn("Unknown pet state", zap.Int("state", newState))
		return nil, pkgerrors.ErrInvalidArgs
	}

	if p != nil {
		// Send onUpdatePet callback
		_ = ctx.Connection.SendCallback("onUpdatePet", p.ID, "state", newState)

		// Send onRefreshPetProp callback with computed stats
		petDTO := p.ToDTO()
		if data, ok := petDTO["data"].(map[string]interface{}); ok {
			petDTO = data
		}
		if prop, ok := petDTO["property"]; ok {
			_ = ctx.Connection.SendCallback("onRefreshPetProp", map[string]interface{}{
				"id": p.ID,
				"s":  prop,
			})
		}
	}

	return nil, nil
}

// FinalPraMagDefPet handles client's request to recalculate/refresh pet stats.
// Arguments from log: [{}, "finalPraMagDef", 0, charID]
func (h *PetHandler) FinalPraMagDefPet(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	h.logger.Debug("FinalPraMagDefPet called", zap.Int64("character_id", characterID))

	pets, err := h.petService.GetPetList(ctx.Context, characterID)
	if err != nil {
		return rtmp.ErrorToResponse(err), nil
	}

	for _, p := range pets {
		p.RecalculateStats()
		petDTO := p.ToDTO()
		if data, ok := petDTO["data"].(map[string]interface{}); ok {
			petDTO = data
		}
		if prop, ok := petDTO["property"]; ok {
			_ = ctx.Connection.SendCallback("onRefreshPetProp", map[string]interface{}{
				"id": p.ID,
				"s":  prop,
			})
		}
	}

	return nil, nil
}

// DelPetByClient handles client's request to delete a pet.
// Arguments: [null, petID, password] or [petID, password]
func (h *PetHandler) DelPetByClient(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	if len(args) < 1 {
		return nil, pkgerrors.ErrInvalidArgs
	}

	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	var petID int64
	var argIndex int

	if args[0] == nil {
		if len(args) < 2 {
			return nil, pkgerrors.ErrInvalidArgs
		}
		argIndex = 1
	} else {
		argIndex = 0
	}

	switch v := args[argIndex].(type) {
	case float64:
		petID = int64(v)
	case int64:
		petID = v
	case int:
		petID = int64(v)
	default:
		h.logger.Warn("DelPetByClient: invalid petID type", zap.Any("arg", args[argIndex]), zap.String("type", fmt.Sprintf("%T", args[argIndex])))
		return nil, pkgerrors.ErrInvalidArgs
	}

	// Password validation
	if argIndex+1 < len(args) {
		if password, ok := args[argIndex+1].(string); ok {
			accountID, err := uuid.Parse(ctx.AccountID)
			if err != nil {
				h.logger.Error("DelPetByClient: invalid accountID", zap.String("accountID", ctx.AccountID), zap.Error(err))
				return nil, pkgerrors.ErrUnauthorized
			}

			account, err := h.accountRepo.FindByID(ctx.Context, accountID)
			if err != nil {
				h.logger.Error("DelPetByClient: account not found", zap.String("accountID", ctx.AccountID), zap.Error(err))
				return nil, pkgerrors.ErrAccountNotFound
			}

			if !account.CheckSecondaryPassword(password) {
				h.logger.Warn("DelPetByClient: incorrect secondary password", zap.String("accountID", ctx.AccountID))
				return nil, fmt.Errorf("Incorrect secondary password")
			}
		}
	}

	h.logger.Info("Deleting pet",
		zap.Int64("character_id", characterID),
		zap.Int64("pet_id", petID))

	if err := h.petService.DeletePet(ctx.Context, characterID, petID); err != nil {
		h.logger.Error("Failed to delete pet", zap.Error(err))
		return rtmp.ErrorToResponse(err), nil
	}

	// Send callback to client
	if err := ctx.Connection.SendCallback("onDelPet", petID); err != nil {
		h.logger.Warn("Failed to send onDelPet callback", zap.Error(err))
	}

	return map[string]interface{}{"success": true}, nil
}
