// Social handler for friend and relationship RPC calls.
// Handles friend list, blacklist, and IM panel initialization.
// Returns relationship data with online status for client UI.
package handlers

import (
	"strconv"

	appsocial "mcgame-server/internal/application/social"
	"mcgame-server/internal/infrastructure/rtmp"
	pkgerrors "mcgame-server/pkg/errors"

	"go.uber.org/zap"
)

type SocialHandler struct {
	socialService *appsocial.Service
	logger        *zap.Logger
}

func NewSocialHandler(socialService *appsocial.Service, logger *zap.Logger) *SocialHandler {
	return &SocialHandler{
		socialService: socialService,
		logger:        logger,
	}
}

func (h *SocialHandler) RegisterHandlers(dispatcher *rtmp.RPCDispatcher) {
	dispatcher.Register("initViewImC", h.InitViewImC)
	dispatcher.Register("getFriendList", h.GetFriendList)
	dispatcher.Register("getBlackList", h.GetBlackList)
	dispatcher.Register("addRelationByName", h.AddRelationByName)
	dispatcher.Register("delRelationship", h.DelRelationship)
	dispatcher.Register("initTS", h.InitTS)
	dispatcher.Register("findTeacher", h.FindTeacher)
	dispatcher.Register("findStudent", h.FindStudent)
	dispatcher.Register("reportTS", h.ReportTS)
	dispatcher.Register("delST", h.DelST)
}

func (h *SocialHandler) InitViewImC(ctx *rtmp.RPCContext, args []any) (any, error) {
	charIDStr := ctx.CharacterID
	if charIDStr == "" {
		h.logger.Warn("InitViewImC: no character ID in session",
			zap.Uint32("conn_id", ctx.ConnID))
		return nil, pkgerrors.ErrUnauthorized
	}

	characterID, err := strconv.ParseInt(charIDStr, 10, 64)
	if err != nil {
		h.logger.Warn("InitViewImC: invalid character ID",
			zap.String("char_id", charIDStr),
			zap.Error(err))
		return nil, pkgerrors.ErrInvalidInput
	}

	h.logger.Debug("InitViewImC: fetching relationship list",
		zap.Uint32("conn_id", ctx.ConnID),
		zap.Int64("character_id", characterID))

	relationships, err := h.socialService.GetRelationshipList(ctx.Context, characterID)
	if err != nil {
		h.logger.Error("InitViewImC: failed to get relationship list",
			zap.Int64("character_id", characterID),
			zap.Error(err))
		return nil, err
	}

	result := make(map[string]any)
	for _, rel := range relationships {
		relID := rel["id"]
		result[strconv.FormatInt(relID.(int64), 10)] = rel
	}

	h.logger.Info("InitViewImC: returning relationship list",
		zap.Uint32("conn_id", ctx.ConnID),
		zap.Int64("character_id", characterID),
		zap.Int("count", len(result)))

	if err := ctx.Connection.SendCallback("onInitViewImC", result); err != nil {
		h.logger.Error("InitViewImC: failed to send callback",
			zap.Uint32("conn_id", ctx.ConnID),
			zap.Error(err))
	}

	return nil, nil
}

// AddRelationByName adds a relationship (friend, blacklist, enemy, etc.) by character name.
// Client call: _core.remote.addRelationByName(name, type)
// Parameters:
//   - name (String): Target character's name
//   - type (Number): Relationship type (0=friend, 1=blacklist, 2=enemy, 3=couple, 4=tutor, 5=brother)
//
// Response: Sends callback onAddRelationByName(characterId, type)
//   - If characterId = -1, player not found
//   - Otherwise, relationship was added successfully
func (h *SocialHandler) AddRelationByName(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	if len(args) < 2 {
		return nil, pkgerrors.ErrInvalidArgs
	}

	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {
		return nil, pkgerrors.ErrUnauthorized
	}

	// Parse name
	name, ok := args[0].(string)
	if !ok {
		h.logger.Warn("AddRelationByName: invalid name type",
			zap.Any("name", args[0]))
		return nil, pkgerrors.ErrInvalidInput
	}

	// Parse relationship type
	var relType int
	switch v := args[1].(type) {
	case float64:
		relType = int(v)
	case int:
		relType = v
	default:
		h.logger.Warn("AddRelationByName: invalid type",
			zap.Any("type", args[1]))
		return nil, pkgerrors.ErrInvalidInput
	}

	h.logger.Info("AddRelationByName called",
		zap.Uint32("conn_id", ctx.ConnID),
		zap.Int64("character_id", characterID),
		zap.String("target_name", name),
		zap.Int("relationship_type", relType))

	// Add relationship using service
	relDTO, err := h.socialService.AddRelationshipByName(ctx.Context, characterID, name, relType)
	if err != nil {
		h.logger.Warn("AddRelationByName: failed to add relationship",
			zap.String("target_name", name),
			zap.Int("relationship_type", relType),
			zap.Error(err))

		// Send callback with -1 to indicate failure (player not found or error)
		if err := ctx.Connection.SendCallback("onAddRelationByName", -1, relType); err != nil {
			h.logger.Error("Failed to send onAddRelationByName error callback",
				zap.Uint32("conn_id", ctx.ConnID),
				zap.Error(err))
		}

		// Return error response
		return map[string]interface{}{
			"success": false,
			"error":   err.Error(),
		}, nil
	}

	// Get target character ID from relationship DTO
	targetID, ok := relDTO["otherId"].(int64)
	if !ok {
		targetID = -1
	}

	// Send success callback
	if err := ctx.Connection.SendCallback("onAddRelationByName", targetID, relType); err != nil {
		h.logger.Error("Failed to send onAddRelationByName callback",
			zap.Uint32("conn_id", ctx.ConnID),
			zap.Error(err))
	}

	h.logger.Info("Relationship added",
		zap.Int64("character_id", characterID),
		zap.Int64("target_id", targetID),
		zap.String("target_name", name),
		zap.Int("relationship_type", relType))

	return map[string]interface{}{
		"success":      true,
		"relationship": relDTO,
	}, nil
}

// DelRelationship deletes a relationship by its ID.

		return nil, pkgerrors.ErrInvalidArgs
	}

	characterID, err := strconv.ParseInt(ctx.CharacterID, 10, 64)
	if err != nil {

	}

	// Parse relationship ID
	var relID int64
	switch v := args[0].(type) {
	case float64:
		relID = int64(v)
	case int:

	case int64:
		relID = v
	default:
		h.logger.Warn("DelRelationship: invalid relationship ID type",
			zap.Any("rel_id", args[0]))
		return nil, pkgerrors.ErrInvalidInput
	}

	h.logger.Info("DelRelationship called",
		zap.Uint32("conn_id", ctx.ConnID),

	h.logger.Info("Relationship deleted",
		zap.Int64("character_id", characterID),
		zap.Int64("relationship_id", relID))

	return nil, nil
}

// InitTS initializes the teacher/student view.
// Client call: _core.remote.initTS()
//
// Response: Sends callback onInitTS({students[]})
func (h *SocialHandler) InitTS(ctx *rtmp.RPCContext, args []interface{}) (interface{}, error) {
	charIDStr := ctx.CharacterID
	if charIDStr == "" {
		h.logger.Warn("InitTS: no character ID in session",
			zap.Uint32("conn_id", ctx.ConnID))
		return nil, pkgerrors.ErrUnauthorized
	}

	characterID, err := strconv.ParseInt(charIDStr, 10, 64)
	if err != nil {
		h.logger.Warn("InitTS: invalid character ID",
			zap.String("char_id", charIDStr),
			zap.Error(err))
		return nil, pkgerrors.ErrInvalidInput
	}

	h.logger.Debug("InitTS: fetching teacher/student list",
		zap.Uint32("conn_id", ctx.ConnID),
		zap.Int64("character_id", characterID))

	students, err := h.socialService.GetTeacherStudentList(ctx.Context, characterID)
	if err != nil {
		h.logger.Error("InitTS: failed to get teacher/student list",
			zap.Int64("character_id", characterID),
			zap.Error(err))
		return nil, err

	for _, student := range students {
		studentID := student["id"]
		result[strconv.FormatInt(studentID.(int64), 10)] = student

	h.logger.Info("InitTS: returning teacher/student list",
		zap.Uint32("conn_id", ctx.ConnID),

		h.logger.Error("InitTS: failed to send callback",
			zap.Uint32("conn_id", ctx.ConnID),
			zap.Error(err))
	}

	return nil, nil
}
